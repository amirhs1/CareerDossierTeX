% /Lang(fr-CA) in the catalog can only have come from the key.

  name={Metadata Fixture}, email={metadata@example.test}, language={fr-CA}
}

\begin{document}
\MakeCDossierLetterhead
A paragraph, so the letter has a body.
\MakeCDossierClosing
\end{document}
