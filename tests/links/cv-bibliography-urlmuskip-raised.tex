% cv-bibliography-urlmuskip-raised.tex — the negative control for issue #294,
% made toolchain-independent by issue #312.
%
% Identical to cv-bibliography-links.tex except that it raises \biburlbigskip,
% which \biburlsetup copies into \Urlmuskip, so every breakpoint inside a URL
% gains measurable width. Current Poppler starts a new word when the spacing
% between two characters exceeds 0.1x the font size, so the URL extracts as
% `https : / / records . example . invalid /' — the failure #199 found.
%
% This fixture must therefore FAIL the token check, and run.sh asserts exactly
% that. Without it the suite could pass because it looks in the wrong place, or
% because the checker cannot see the defect at all.
%
% The glue is a rigid 4mu, 2/9 em and more than twice the threshold. Do not
% replace it with BibLaTeX's own 0mu plus 3mu, which this fixture used until
% #312: a *stretch* only reaches the extractor's threshold when the line it
% lands on happens to need enough stretch, and the realized per-breakpoint gap
% then sits within a point or two of 0.1 em. That is not an assertion, it is a
% coin toss. Rigid, the gap is decided by this file, not by the toolchain. The
% measurements behind this, and what the control gives up for it, are in #312
% and in its pull request, #313.
%
% LINKEXPECT: split
% LINKTOKEN: https://records.example.invalid/collections/analytical-engine/notes/1843/appendix-g.html
% LINKTOKEN: https://en.wikipedia.org/w/index.php?search=research-creation+sound-producing+Iranian+adornments+museum+conservation+archival+silencing+diasporic+futurism+embodied+performance+sound+studies+material+culture+museum+collections+classification+preservation&title=Special%3ASearch&profile=advanced&fulltext=1&ns0=1
\documentclass{careerdossier-cv}
\usepackage{careerdossier-biblatex}
\addbibresource{links.bib}
\CDossierSetup{name={Ada Lovelace}}
\biburlbigskip=4mu % widen every URL breakpoint past 0.1em; see above
\begin{document}
\nocite{*}
\printbibliography[title={Publications}]
\end{document}
