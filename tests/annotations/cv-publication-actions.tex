% cv-publication-actions.tex — the manual publication list's identifier links
% (#328).
%
% careerdossier-cv normalizes a DOI or a fallback URL with its own copy of the
% scheme insertion, so it carried the same defect at three sites: the bare DOI
% (whole `https://doi.org/' prefix), the `doi.org/' spelling (scheme only), and
% the fallback URL (scheme only). All six entries below must produce URI
% actions, and each written-out spelling must match its short form exactly.
%
% A DOI written with a `doi:' prefix must link to the same address as the bare
% identifier rather than to `https://doi.org/doi:...' (#688).
%
% A fallback URL that carries a scheme of its own keeps it (#740): a link
% value carries a scheme when it contains `://' or begins with `mailto:' or
% `tel:', the rule careerdossier-links applies to every link value. Before the
% fix the `mailto:' entry linked to `https://mailto:pub@example.test'. A port
% number is not a scheme, so `example.invalid:8080/x' still gains `https://';
% a test for any `letters:' prefix would read `example.invalid' as one. The
% scheme-less fallback URL above is the case that still gains `https://'.
%
% Addresses are kept short so no entry's identifier wraps; see the note in
% resume-contact-actions.tex on why a wrap would confuse the comparison.
%
% URIEXPECT: mailto:ada@example.test
% URIEXPECT: https://doi.org/10.9999/bare
% URIEXPECT: https://doi.org/10.9999/host
% URIEXPECT: https://doi.org/10.9999/full
% URIEXPECT: https://doi.org/10.9999/prefixed
% URIEXPECT: https://example.invalid/note
% URIEXPECT: https://example.invalid/full
% URIEXPECT: mailto:pub@example.test
% URIEXPECT: https://example.invalid:8080/x
\documentclass{careerdossier-cv}
\input{uncompressed.inc}
\CDossierSetup{
  name  = {Ada Lovelace},
  email = {ada@example.test},
}
\begin{document}
\MakeCDossierHeader
\CDossierSection{Publications}
\begin{CDossierPublications}
  \CDossierPublication{
    authors = {Ada Lovelace},
    title   = {A Bare DOI},
    doi     = {10.9999/bare}
  }
  \CDossierPublication{
    authors = {Ada Lovelace},
    title   = {A DOI Naming Its Host},
    doi     = {doi.org/10.9999/host}
  }
  \CDossierPublication{
    authors = {Ada Lovelace},
    title   = {A Written-Out DOI},
    doi     = {https://doi.org/10.9999/full}
  }
  \CDossierPublication{
    authors = {Ada Lovelace},
    title   = {A Prefixed DOI},
    doi     = {doi:10.9999/prefixed}
  }
  \CDossierPublication{
    authors = {Ada Lovelace},
    title   = {A Scheme-Less Fallback URL},
    url     = {example.invalid/note}
  }
  \CDossierPublication{
    authors = {Ada Lovelace},
    title   = {A Written-Out Fallback URL},
    url     = {https://example.invalid/full}
  }
  \CDossierPublication{
    authors = {Ada Lovelace},
    title   = {A Mail Fallback URL},
    url     = {mailto:pub@example.test}
  }
  \CDossierPublication{
    authors = {Ada Lovelace},
    title   = {A Fallback URL With a Port},
    url     = {example.invalid:8080/x}
  }
\end{CDossierPublications}
\end{document}
