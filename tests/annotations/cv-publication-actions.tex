% cv-publication-actions.tex — the manual publication list's identifier links
% (issue #328).
%
% careerdossier-cv normalizes a DOI or a fallback URL with its own copy of the
% scheme insertion, so it carried the same defect at three sites: the bare DOI
% (whole `https://doi.org/' prefix), the `doi.org/' spelling (scheme only), and
% the fallback URL (scheme only). All five entries below must produce URI
% actions, and each written-out spelling must match its short form exactly.
%
% Addresses are kept short so no entry's identifier wraps; see the note in
% resume-contact-actions.tex on why a wrap would confuse the comparison.
%
% URIEXPECT: mailto:ada@example.test
% URIEXPECT: https://doi.org/10.9999/bare
% URIEXPECT: https://doi.org/10.9999/host
% URIEXPECT: https://doi.org/10.9999/full
% URIEXPECT: https://example.invalid/note
% URIEXPECT: https://example.invalid/full
\documentclass{careerdossier-cv}
\input{uncompressed.inc}
\CDossierSetup{
  name  = {Ada Lovelace},
  email = {ada@example.test},
}
\begin{document}
\MakeCDossierHeader
\CDossierSection{Publications}
\begin{CDossierPublications}
  \CDossierPublication{
    authors = {Ada Lovelace},
    title   = {A Bare DOI},
    doi     = {10.9999/bare}
  }
  \CDossierPublication{
    authors = {Ada Lovelace},
    title   = {A DOI Naming Its Host},
    doi     = {doi.org/10.9999/host}
  }
  \CDossierPublication{
    authors = {Ada Lovelace},
    title   = {A Written-Out DOI},
    doi     = {https://doi.org/10.9999/full}
  }
  \CDossierPublication{
    authors = {Ada Lovelace},
    title   = {A Scheme-Less Fallback URL},
    url     = {example.invalid/note}
  }
  \CDossierPublication{
    authors = {Ada Lovelace},
    title   = {A Written-Out Fallback URL},
    url     = {https://example.invalid/full}
  }
\end{CDossierPublications}
\end{document}
