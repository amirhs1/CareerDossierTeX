% cv-publications-valid.tex — dependency-free manual publications. The class
% must not make BibLaTeX commands available as a side effect.
\documentclass{careerdossier-cv}
\CDossierSetup{name={Ada Lovelace}}

\msg_new:nnn { cdossier-test } { unexpected-biblatex-dependency }
  { The~CV~class~unexpectedly~loaded~BibLaTeX. }
\AtBeginDocument
  {
    \cs_if_exist:NT \printbibliography
      { \msg_error:nn { cdossier-test } { unexpected-biblatex-dependency } }
  }

  }

\end{document}
