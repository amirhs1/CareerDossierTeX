% resume-section-leading.tex — the section paragraph must end while the
% calibrated section size is active, so TeX contributes section leading rather
% than body leading to the vertical list.
\documentclass{careerdossier-resume}
\CDossierSetup{name = {Ada Lovelace}}
\ExplSyntaxOn
\bool_new:N \g__cdossier_test_section_par_checked_bool
\cs_new_eq:NN \__cdossier_test_saved_par: \par
\msg_new:nnn { cdossier-test } { wrong-section-leading }
  { Section~paragraph~ended~at~\dim_use:N \baselineskip;~expected~14pt. }
\cs_set_protected:Npn \__cdossier_typography_heading:nn #1#2
  {
    \group_begin:
      \cs_set_protected:Npn \par
        {
          \bool_if:NF \g__cdossier_test_section_par_checked_bool
            {
              \dim_compare:nNnF { \baselineskip } = { 14pt }
                { \msg_error:nn { cdossier-test } { wrong-section-leading } }
              \bool_gset_true:N \g__cdossier_test_section_par_checked_bool
            }
          \__cdossier_test_saved_par:
        }
      \noindent #2
      \par
    \group_end:
  }
\ExplSyntaxOff
\begin{document}
\CDossierSection{Experience}
\ExplSyntaxOn
%% A macro-name rename that stops this stub from ever firing must fail loudly
%% rather than pass by never running its check (issue #267 renamed the
%% patched primitive from :n to :nn).
\bool_if:NF \g__cdossier_test_section_par_checked_bool
  { \msg_error:nn { cdossier-test } { wrong-section-leading } }
\ExplSyntaxOff
\end{document}
