% examples/academic/cv-academic.tex
%
% Supported v0.2 academic-CV example with dependency-free manual publications.
% It deliberately has no bibliography package; the external BibLaTeX/Biber
% integration remains optional and is demonstrated separately.
%
% Compile with LuaLaTeX:
%   latexmk -lualatex -interaction=nonstopmode -halt-on-error \
%     examples/academic/cv-academic.tex

\documentclass[fontsize=12pt, margin=normal]{careerdossier-cv}
\input{examples/profiles/profile-academic.tex}

\begin{document}

\MakeCDossierHeader

\CDossierSection{Academic Appointments}
\begin{CDossierEntry}[
  title        = {Research Fellow},
  organization = {Example Institute for Analytical Computing},
  location     = {Toronto, Ontario},
  dates        = {2024--Present}
]
  \begin{CDossierItemize}
    \item Develop reproducible methods for scientific computing and data-driven
          modelling.
    \item Supervise student research projects and maintain open teaching
          materials.
  \end{CDossierItemize}
\end{CDossierEntry}

\CDossierSection{Education}
\begin{CDossierEntry}[
  title        = {Ph.D. in Computational Science},
  organization = {Example University},
  location     = {Toronto, Ontario},
  dates        = {2018--2024}
]
  Dissertation: Reliable numerical methods for coupled nonlinear systems.
\end{CDossierEntry}

% \CDossierSubsection is the level between a section and an entry. Grouping a
% section this way keeps one ruled heading for the section and marks the groups
% under it, instead of promoting each group to a section of its own — which
% would flatten exactly the hierarchy the grouping exists to show.
\CDossierSection{Teaching}
\CDossierSubsection{Course Instruction}
\begin{CDossierEntry}[
  title        = {Sessional Instructor},
  organization = {MATH 2410: Differential Equations},
  location     = {Example University},
  dates        = {Winter 2025}
]
  Designed accessible assessments and supported a diverse cohort of students.
\end{CDossierEntry}

\CDossierSubsection{Graduate Supervision}
\begin{CDossierEntry}[
  title        = {Co-supervisor, M.Sc. Thesis},
  organization = {Example University},
  location     = {Toronto, Ontario},
  dates        = {2025--Present}
]
  Numerical stability of coupled solvers under sparse observational data.
\end{CDossierEntry}

% The same command groups a publication list, which is where an academic CV
% needs it most. The gap under the heading is the same in both cases: one pair
% of tokens serves an entry run and a CDossierPublications list alike.

\CDossierSubsection{Journal Articles}

\end{document}
